% Dice rendering
\newcommand{\TableDiceSymbol}[1]{%
  \ifnum#1<1
    \PackageError{dictionary}{Dice value #1 is below 1}{}
  \else
    \ifnum#1>6
      \PackageError{dictionary}{Dice value #1 is above 6}{}
    \else
      \char\numexpr"F08A+#1\relax
    \fi
  \fi}
